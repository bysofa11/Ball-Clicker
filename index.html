// VARIABLES let clicks = 0; let multiplier = 1; let clickValue = 1; let clickUpgradeLevel = 0; let clickUpgradeCost = 50; const CLICK_UPGRADE_GROWTH = 1.15; let bootLevel = 0; let bootCost = 200; const BOOT_GROWTH = 1.2; // ELEMENTOS const ball = document.getElementById("ball"); const bootOrbit = document.getElementById("bootOrbit"); const clicksText = document.getElementById("clicks"); const shopBallsCount = document.getElementById("shopBallsCount"); const shopButton = document.getElementById("shopButton"); const shop = document.getElementById("shop"); const closeShop = document.getElementById("closeShop"); const buyDouble = document.getElementById("buyDouble"); const buyClickUpgrade = document.getElementById("buyClickUpgrade"); const clickUpgradeLevelText = document.getElementById("clickUpgradeLevel"); const clickUpgradeCostText = document.getElementById("clickUpgradeCostText"); const buyBoot = document.getElementById("buyBoot"); const bootLevelText = document.getElementById("bootLevel"); const bootCostText = document.getElementById("bootCostText"); const message = document.getElementById("message"); // SONIDO DE CLICK (generado con Web Audio API, sin archivos externos) let audioCtx; function getAudioCtx() { if (!audioCtx) { audioCtx = new (window.AudioContext || window.webkitAudioContext)(); } if (audioCtx.state === "suspended") { audioCtx.resume(); } return audioCtx; } function playClickSound() { const ctx = getAudioCtx(); const osc = ctx.createOscillator(); const gain = ctx.createGain(); osc.type = "sine"; osc.frequency.setValueAtTime(650, ctx.currentTime); osc.frequency.exponentialRampToValueAtTime(220, ctx.currentTime + 0.1); gain.gain.setValueAtTime(0.2, ctx.currentTime); gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.12); osc.connect(gain).connect(ctx.destination); osc.start(); osc.stop(ctx.currentTime + 0.12); } // ANIMACIÓN DE CLICK / GOLPE function playClickAnimation() { ball.classList.remove("clicked"); void ball.offsetWidth; // fuerza reinicio de la animación ball.classList.add("clicked"); } function playHitAnimation() { ball.classList.remove("hit"); void ball.offsetWidth; ball.classList.add("hit"); } // CLICK EN LA PELOTA ball.addEventListener("click", function() { clicks += multiplier; clicks += clickValue; clicksText.textContent = clicks; refreshUI(); playClickSound(); playClickAnimation(); }); }); // COMPRAR X2 // ACTUALIZAR TEXTOS DE LA TIENDA Y BOTONES buyDouble.addEventListener("click", function() { function updateShopUI() { const price = 50; clickUpgradeLevelText.textContent = clickUpgradeLevel; clickUpgradeCostText.textContent = clickUpgradeCost; if (clicks >= price) { bootLevelText.textContent = bootLevel; bootCostText.textContent = bootCost; clicks -= price; buyClickUpgrade.classList.toggle("insufficient", clicks < clickUpgradeCost); buyBoot.classList.toggle("insufficient", clicks < bootCost); multiplier = 2; } function refreshUI() { clicksText.textContent = clicks; clicksText.textContent = clicks; shopBallsCount.textContent = clicks; message.textContent = "✅ ¡Mejora comprada! Ahora cada click vale x2."; updateShopUI(); } buyDouble.disabled = true; buyDouble.textContent = "Comprado"; // COMPRAR +1 POR CLICK (infinito, precio creciente) buyClickUpgrade.addEventListener("click", function() { if (clicks >= clickUpgradeCost) { clicks -= clickUpgradeCost; clickValue += 1; clickUpgradeLevel += 1; clickUpgradeCost = Math.ceil(clickUpgradeCost * CLICK_UPGRADE_GROWTH); refreshUI(); message.textContent = "✅ ¡Mejora comprada! Ahora cada click vale +" + clickValue + " pelotas."; } else { message.textContent = "❌ No tienes suficientes clicks."; message.textContent = "❌ No tienes suficientes pelotas."; } }); // COMPRAR BOTÍN CON TACOS (infinito, precio creciente) buyBoot.addEventListener("click", function() { if (clicks >= bootCost) { clicks -= bootCost; bootLevel += 1; bootCost = Math.ceil(bootCost * BOOT_GROWTH); refreshUI(); renderBoots(); message.textContent = "✅ ¡Botín añadido!
